FORM update_device_locations.

    lv_tplnr         TYPE bapi_itob_parms-funcloc_int,
    ls_general       TYPE bapi_itob,
    ls_generalx      TYPE bapi_itobx,
    ls_specific      TYPE bapi_itob_fl_only,
    ls_specificx     TYPE bapi_itob_fl_onlyx,
    ls_general_exp   TYPE bapi_itob,
    ls_specific_exp  TYPE bapi_itob_fl_only,
    ls_return        TYPE bapiret2,
    lv_count         TYPE i.
*    lv_anlage        TYPE eastl-anlage,
*    lv_mprn          TYPE euitrans-ext_ui,

LOOP AT gt_upload INTO gs_upload.

  CLEAR:
    lv_tplnr,
    ls_return.

**--------------------------------------------------------------------*
** Find Current MPRN
**--------------------------------------------------------------------*
SELECT SINGLE f~tplnr
  INTO @lv_tplnr
  FROM egerh     AS h
  INNER JOIN eastl     AS ea ON ea~logiknr = h~logiknr
  INNER JOIN euiinstln AS ei ON ei~anlage  = ea~anlage
  INNER JOIN euitrans  AS et ON et~int_ui  = ei~int_ui
  INNER JOIN iflot     AS f  ON f~tplnr    = h~devloc
  WHERE et~ext_ui = @gs_upload-mprn
    AND h~bis     = '99991231'.

   IF sy-subrc <> 0.
    gs_upload-status  = 'ERROR'.
    gs_upload-message = 'Functional location not found'.
    MODIFY gt_upload FROM gs_upload.
    CONTINUE.

  ENDIF.
****************
* Retaining this code if want to find from Serial Num
****************
** Find current device location
*  SELECT SINGLE h~devloc
*    INTO @lv_tplnr
*    FROM equi AS e
*    INNER JOIN egerh AS h
*      ON h~equnr = e~equnr
*   WHERE e~sernr = @gs_upload-sernr
*     AND h~bis   = '99991231'.
*
*  IF sy-subrc <> 0.
*    gs_upload-status  = 'ERROR'.
*    gs_upload-message = 'Serial number not found'.
*    MODIFY gt_upload FROM gs_upload.
*    CONTINUE.
*
*  ENDIF.
*
**--------------------------------------------------------------------*
** Find Current MPRN
**--------------------------------------------------------------------*
*    SELECT SINGLE l~anlage
*      INTO @lv_anlage
*      FROM equi AS e
*      INNER JOIN egerh AS h
*        ON h~equnr = e~equnr
*      INNER JOIN eastl AS l
*        ON l~logiknr = h~logiknr
*     WHERE e~sernr = @gs_upload-sernr
*       AND h~bis   = '99991231'.
*
*    SELECT SINGLE t~ext_ui
*      INTO @lv_mprn
*      FROM EUIINSTLN AS i
*      INNER JOIN euitrans AS t
*        ON t~int_ui = i~int_ui
*     WHERE i~anlage = @lv_anlage
*       AND i~dateto = '99991231'

*      gs_upload-status  = 'ERROR'.

*      MODIFY gt_upload FROM gs_upload.

  CLEAR: ls_general,
         ls_generalx.

  CASE gs_upload-location.

    WHEN 'INDOOR'.
      ls_general-maintloc = 'INDOOR'.

    WHEN 'OUTDOOR'.
      ls_general-maintloc = 'OUTDOOR'.

    WHEN OTHERS.

        EXPORTING

FORM get_outside_meter_locations .

  DATA:
    lv_next_year TYPE gjahr,
    lv_mprn          TYPE euitrans-ext_ui,
    lv_anlage        TYPE eastl-anlage.

  CLEAR gt_report.

  IF p_addr IS INITIAL.

  SELECT
      egerh~devloc,
      equi~sernr,
      equi~matnr,
      iloa~stort,
      eastl~anlage,
"      iloa~stortzus,
      egers~bgljahr,
      etyp~vlzeitti,
      adrc~post_code1,
      adrc~house_num1,
      adrc~street,
      adrc~city1,
      euitrans~ext_ui,
      eastl~logiknr
    INTO TABLE @DATA(lt_data)

    FROM iloa

    INNER JOIN adrc
ON adrc~addrnumber = iloa~ADRNR
    INNER JOIN iflot
      ON iflot~iloan = iloa~iloan
    INNER JOIN egerh
      ON egerh~devloc = iflot~TPLNR
      INNER JOIN eastl
      ON eastl~LOGIKNR = egerh~LOGIKNR
    INNER JOIN equi
      ON equi~equnr = egerh~equnr
    INNER JOIN egers
      ON egers~equnr = egerh~equnr
    INNER JOIN etyp
      ON etyp~matnr = equi~matnr
    INNER JOIN EUIINSTLN
      ON EUIINSTLN~anlage = eastl~anlage
      INNER JOIN EUITRANS
      ON EUITRANS~int_ui = EUIINSTLN~int_ui

   WHERE iloa~stort IN @s_mloc
    AND egerh~bis = '99991231'
    AND eastl~anlage IN @s_anlage
    AND adrc~post_code1 IN @s_postco
    AND adrc~house_num1 IN @s_hsnr.

  ELSE.

    "Never execute FOR ALL ENTRIES with an empty input table.
    IF gt_addresses IS INITIAL.
      MESSAGE 'No uploaded addresses available'
        TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

  SELECT
      egerh~devloc,
      equi~sernr,
      equi~matnr,
      iloa~stort,
      eastl~anlage,
"      iloa~stortzus,
      egers~bgljahr,
      etyp~vlzeitti,
      adrc~post_code1,
      adrc~house_num1,
      adrc~street,
      adrc~city1,
      euitrans~ext_ui,
      eastl~logiknr
    INTO TABLE @lt_data

    FROM iloa

    INNER JOIN adrc
ON adrc~addrnumber = iloa~ADRNR
    INNER JOIN iflot
      ON iflot~iloan = iloa~iloan
    INNER JOIN egerh
      ON egerh~devloc = iflot~TPLNR
      INNER JOIN eastl
      ON eastl~LOGIKNR = egerh~LOGIKNR
    INNER JOIN equi
      ON equi~equnr = egerh~equnr
    INNER JOIN egers
      ON egers~equnr = egerh~equnr
    INNER JOIN etyp
      ON etyp~matnr = equi~matnr
    INNER JOIN EUIINSTLN
      ON EUIINSTLN~anlage = eastl~anlage
      INNER JOIN EUITRANS
      ON EUITRANS~int_ui = EUIINSTLN~int_ui

    FOR ALL ENTRIES IN @gt_addresses
    WHERE egerh~bis = '99991231'
      AND adrc~house_num1 = @gt_addresses-house_num1
      AND adrc~post_code1 = @gt_addresses-post_code1.

  ENDIF.

    SELECT SINGLE t~ext_ui
      INTO @lv_mprn
      FROM EUIINSTLN AS i
      INNER JOIN euitrans AS t
        ON t~int_ui = i~int_ui
     WHERE i~anlage = @lv_anlage
       AND i~dateto = '99991231'
       AND t~dateto = '99991231'.

  LOOP AT lt_data ASSIGNING FIELD-SYMBOL(<ls_data>).

    CLEAR gs_report.
    gs_report-devloc      = <ls_data>-devloc.
    gs_report-ext_ui      = <ls_data>-ext_ui.
    gs_report-anlage      = <ls_data>-anlage.
    gs_report-sernr       = <ls_data>-sernr.
    gs_report-matnr       = <ls_data>-matnr.
    gs_report-stort       = <ls_data>-stort.
"    gs_report-stortzus    = <ls_data>-stortzus.
    gs_report-bgljahr     = <ls_data>-bgljahr.
    gs_report-vlzeitti    = <ls_data>-vlzeitti.
    gs_report-house_num1 = <ls_data>-house_num1.
    gs_report-address = | { <ls_data>-street }, { <ls_data>-city1 }|.
    gs_report-POST_CODE1    = <ls_data>-POST_CODE1.
    gs_report-next_repyear =     gs_report-bgljahr + gs_report-vlzeitti.
    gs_report-logiknr     = <ls_data>-logiknr.

          IF p_addr = 'X' OR gs_report-next_repyear IN s_repyr.
            APPEND gs_report TO gt_report.
          ENDIF.

  ENDLOOP.

ENDFORM.

FORM upload_file.

  DATA:
    lt_raw  TYPE TABLE OF string,
    lv_line TYPE string,
    lv_filename TYPE string.

  lv_filename = p_csvfil.

      filetype = 'ASC'
*      CHECK_BOM = 'X'
      codepage = '4110'

  LOOP AT lt_raw INTO lv_line.

************************************************************************
* Address lookup only: upload House Number / Postcode CSV
* Save from Excel as CSV UTF-8 (comma delimited).
* Required header: House Number,Postcode
* This FORM never updates meter locations.
************************************************************************
FORM upload_address_csv.

  DATA:
    lt_raw       TYPE STANDARD TABLE OF string,
    lt_cells     TYPE STANDARD TABLE OF string,
    lt_addresses TYPE STANDARD TABLE OF ty_address_input,
    ls_address   TYPE ty_address_input,
    lv_filename  TYPE string,
    lv_line      TYPE string,
    lv_house     TYPE string,
    lv_postcode  TYPE string,
    lv_message   TYPE string,
    lv_row       TYPE i,
    lv_house_len TYPE i,
    lv_post_len  TYPE i,
    lv_length    TYPE i,
    lv_last      TYPE i,
    lv_header    TYPE abap_bool.

  FIELD-SYMBOLS <lv_cell> TYPE string.

  CLEAR gt_addresses.
  lv_filename = p_adrfil.

  IF sy-batch = 'X' OR lv_filename IS INITIAL.
    MESSAGE 'Select a local address CSV file in SAP GUI' TYPE 'E'.
  ENDIF.

  CALL FUNCTION 'GUI_UPLOAD'
    EXPORTING
      filename                = lv_filename
      filetype                = 'ASC'
      codepage                = '4110'
    TABLES
      data_tab                = lt_raw
    EXCEPTIONS
      file_open_error         = 1
      file_read_error         = 2
      no_batch                = 3
      gui_refuse_filetransfer  = 4
      invalid_type            = 5
      no_authority            = 6
      OTHERS                  = 7.

  IF sy-subrc <> 0.
    MESSAGE 'Address CSV upload failed; check the file and GUI access'
      TYPE 'E'.
  ENDIF.

  DESCRIBE FIELD ls_address-house_num1
    LENGTH lv_house_len IN CHARACTER MODE.
  DESCRIBE FIELD ls_address-post_code1
    LENGTH lv_post_len IN CHARACTER MODE.

  LOOP AT lt_raw INTO lv_line.
    lv_row = sy-tabix.
    IF lv_line IS INITIAL.
      CONTINUE.
    ENDIF.

    CLEAR: lt_cells, lv_house, lv_postcode, ls_address.
    SPLIT lv_line AT ',' INTO TABLE lt_cells.

    IF lines( lt_cells ) <> 2.
      lv_message =
        |CSV row { lv_row }: expected two comma-separated columns|.
      MESSAGE lv_message TYPE 'E'.
    ENDIF.

    "Allow optional enclosing quotes; embedded commas/quotes are invalid
    "for these two address-key fields.
    LOOP AT lt_cells ASSIGNING <lv_cell>.
      CONDENSE <lv_cell>.
      lv_length = strlen( <lv_cell> ).
      IF lv_length >= 2.
        lv_last = lv_length - 1.
        IF <lv_cell>(1) = '"' AND <lv_cell>+lv_last(1) = '"'.
          lv_length = lv_length - 2.
          <lv_cell> = <lv_cell>+1(lv_length).
        ENDIF.
      ENDIF.
      IF <lv_cell> CS '"'.
        lv_message = |CSV row { lv_row }: invalid quotation marks|.
        MESSAGE lv_message TYPE 'E'.
      ENDIF.
      CONDENSE <lv_cell>.
    ENDLOOP.

    READ TABLE lt_cells INTO lv_house INDEX 1.
    READ TABLE lt_cells INTO lv_postcode INDEX 2.

    IF lv_house IS INITIAL AND lv_postcode IS INITIAL.
      CONTINUE.
    ENDIF.

    IF lv_header IS INITIAL.
      TRANSLATE lv_house TO UPPER CASE.
      TRANSLATE lv_postcode TO UPPER CASE.
      CONDENSE lv_house NO-GAPS.
      CONDENSE lv_postcode NO-GAPS.
      "CP also tolerates a UTF-8 BOM retained before the first heading.
      IF lv_house NP '*HOUSENUMBER' OR lv_postcode <> 'POSTCODE'.
        MESSAGE 'CSV header must be House Number,Postcode' TYPE 'E'.
      ENDIF.
      lv_header = abap_true.
      CONTINUE.
    ENDIF.

    TRANSLATE lv_postcode TO UPPER CASE.

    IF lv_house IS INITIAL OR lv_postcode IS INITIAL.
      lv_message =
        |CSV row { lv_row }: house number and postcode are required|.
      MESSAGE lv_message TYPE 'E'.
    ENDIF.

    IF strlen( lv_house ) > lv_house_len
       OR strlen( lv_postcode ) > lv_post_len.
      lv_message = |CSV row { lv_row }: value exceeds SAP field length|.
      MESSAGE lv_message TYPE 'E'.
    ENDIF.

    ls_address-house_num1 = lv_house.
    ls_address-post_code1 = lv_postcode.
    APPEND ls_address TO lt_addresses.
  ENDLOOP.

  IF lt_addresses IS INITIAL.
    MESSAGE 'CSV contains no address data rows' TYPE 'E'.
  ENDIF.

  gt_addresses = lt_addresses.

ENDFORM.
